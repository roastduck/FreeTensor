123 template <class K, class V>
124 inline bool operator==(const ASTHashMap<K, V> &lhs,
125  const ASTHashMap<K, V> &rhs) {
126  if (lhs.size() != rhs.size()) {
127  return false;
128  }
129  for (auto &&[k, v] : lhs) {
130  if (!rhs.count(k) || rhs.at(k) != v) {
131  return false;
132  }
133  }
134  return true;
135 }
